\begin{abstract}
Let $u_0 \in C^\infty(\T)$ be a smooth, divergence-free initial velocity field on the three-dimensional periodic torus $\T = \R^3 / (2\pi\Z)^3$. We construct the \textbf{Frohmanian Symplectic Tether} $\TF$, a canonical minimal geometric completion of Arnold's Lie--Poisson phase space on $\sdiff^*$. The tether augments the classical skew-symmetric Lie--Poisson bracket $\{F, G\}_{\mathrm{LP}}$ with a degenerate symmetric quadratic projection operator $B(F, G)$ weighted by $|\omega|^2$ and an $L^2$-orthogonal projection $\Pperp$ onto the complement of the velocity field $u = \frac{\delta H}{\delta \omega}$, together with the viscous dissipation bracket $-\nu \mathfrak{D}(F, G)$.

Because $\Pperp u = 0$, the tether satisfies the strict degeneracy identity $B(F, H) \equiv 0$, ensuring that the reversible Hamiltonian sector reproduces the exact unmodified 3D Euler dynamics while viscous dissipation and transverse metric alignment act orthogonally to the energy manifold. We derive the explicit \textbf{Frohmanian Vorticity-Alignment Tether Kernel} $\mathcal{K}_{\mathrm{align}}(x, y)$ and prove its Constantin--Fefferman spherical-shell desingularization under local Lipschitz coherence of the vorticity direction $\xi = \omega / |\omega|$. Stating Hypothesis~B cleanly as the single critical strain bound $\int_0^T \norm{S(t)}_{L^\infty(\T)}\,dt \le C_*(\norm{u_0}_{H^s}, \nu, T)$ for all finite $T < T^*$, we prove our \textbf{Conditional Global Regularity Theorem} establishing $T^*=+\infty$ and smoothness in $C^\infty(\T \times (0, \infty))$ (Clay Statement~B for $t>0$), and record the one-directional Beale--Kato--Majda continuation implication as a corollary. Appendix provides spectral numerical verification on 3D Taylor--Green vortex data.
\end{abstract}

\section{Introduction and Functional Setting}

Consider the incompressible Navier--Stokes equations on $\T = \R^3 / (2\pi\Z)^3$ with kinematic viscosity $\nu > 0$ and zero external force:
\begin{align}
\partial_t u + (u \cdot \nabla) u &= -\nabla p + \nu \Delta u, \qquad (x,t) \in \T \times (0, T), \label{eq:NS_vel}\\
\nabla \cdot u &= 0, \qquad u(\cdot, 0) = u_0, \qquad \int_{\T} u_0(x)\,dx = 0. \label{eq:NS_div}
\end{align}
Taking the curl $\omega = \nabla \times u$ eliminates the pressure and gives the vorticity transport equation:
\begin{equation}\label{eq:vorticity}
\partial_t \omega + (u \cdot \nabla)\omega = (\omega \cdot \nabla)u + \nu \Delta \omega, \qquad \nabla \cdot \omega = 0,
\end{equation}
where $u$ is recovered from $\omega$ via the periodic Biot--Savart convolution $u = \KBS * \omega = \nabla \times (-\Delta)^{-1}\omega$.

By Kato's local well-posedness theorem \cite{Kato1972}, for every divergence-free $u_0 \in H^s(\T)$ with $s > 5/2$, there exists a maximal existence time $T^* = T^*(\norm{u_0}_{H^s}, \nu) \in (0, \infty]$ and a unique classical solution
\[
u \in C([0, T^*); H^s(\T)) \cap C^\infty(\T \times (0, T^*)).
\]
Whether $T^* = +\infty$ holds unconditionally for all $u_0 \in C^\infty(\T)$ is Statement~(B) of the Clay Millennium Prize Problem \cite{Fefferman2000}.

\section{Completion of the Arnold Coadjoint Orbit via $\TF$}

Let $\SDiff$ be the group of volume-preserving diffeomorphisms of $\T$ of Sobolev class $H^s$ ($s > 5/2$), with Lie algebra $\sdiff = \{v \in H^s(\T; \R^3) : \nabla \cdot v = 0\}$. Identifying $\sdiff^*$ with divergence-free vorticity fields via the $L^2$ pairing, the kinetic-energy Hamiltonian is
\begin{equation}\label{eq:Hamiltonian}
H(\omega) = \frac{1}{2} \int_{\T} \abs{u}^2 \, dV = \frac{1}{2} \int_{\T} \omega \cdot (-\Delta)^{-1}\omega \, dV.
\end{equation}

\begin{definition}[Classical Arnold Lie--Poisson Bracket]\label{def:arnold}
For smooth functionals $F, G : \sdiff^* \to \R$, Arnold's Lie--Poisson bracket \cite{Arnold1966} is
\begin{equation}\label{eq:arnold_bracket}
\{F, G\}_{\mathrm{LP}}(\omega) = \int_{\T} \omega \cdot \left( \left(\nabla \times \frac{\delta F}{\delta \omega}\right) \times \left(\nabla \times \frac{\delta G}{\delta \omega}\right) \right) dV.
\end{equation}
\end{definition}

\begin{definition}[Projected Orthogonal Complement $\Pperp$ and Quadratic Tether $B(F, G)$]\label{def:B_tether}
Assume $\norm{u}_{L^2(\T)} > 0$. Let $\Pperp$ denote the $L^2$-orthogonal projection onto the orthogonal complement of the velocity field $u$:
\begin{equation}\label{eq:Pperp}
\Pperp \Phi := \Phi - \frac{\langle \Phi, u \rangle_{L^2(\T)}}{\norm{u}_{L^2(\T)}^2} u.
\end{equation}
For a coupling parameter $\kappa \ge 0$, define the quadratic tether correction by
\begin{equation}\label{eq:B_form}
B(F, G)(\omega) := -\kappa \int_{\T} \abs{\omega}^2 \left( \Pperp \left(\nabla \times \frac{\delta F}{\delta \omega}\right) \cdot \Pperp \left(\nabla \times \frac{\delta G}{\delta \omega}\right) \right) dV.
\end{equation}
\end{definition}

\begin{definition}[The Frohmanian Symplectic Tether $\TF$]\label{def:TF}
The complete metriplectic tether is
\begin{equation}\label{eq:TF_full}
\TF(F, G)(\omega) := \{F, G\}_{\mathrm{LP}}(\omega) + B(F, G)(\omega) - \nu \mathfrak{D}(F, G)(\omega),
\end{equation}
where $\mathfrak{D}(F, G) := \int_{\T} \nabla \frac{\delta F}{\delta \omega} : \nabla \frac{\delta G}{\delta \omega} \, dV$. The evolution of any observable is
\begin{equation}\label{eq:TF_flow}
\frac{d}{dt} F(\omega) = \{F, H\}_{\mathrm{LP}}(\omega) + B(F, H)(\omega) - \nu \mathfrak{D}(F, E_{\mathrm{ens}})(\omega).
\end{equation}
\end{definition}

\begin{lemma}[Strict Degeneracy and Preservation of Classical PDE]\label{lem:degeneracy}
For every admissible smooth functional $F$, 
\begin{equation}\label{eq:B_degeneracy}
B(F, H)(\omega) \equiv 0.
\end{equation}
Consequently, the evolution \eqref{eq:TF_flow} reproduces exactly the unmodified 3D incompressible Navier--Stokes vorticity equation, and the kinetic energy satisfies the standard dissipation identity
\begin{equation}\label{eq:energy_dissipation}
\frac{d}{dt} H(\omega(t)) = -\nu \int_{\T} \abs{\omega}^2 \, dV = -\nu \int_{\T} \abs{\nabla u}^2 \, dV \le 0.
\end{equation}
\end{lemma}

\begin{proof}
By definition, $\nabla \times \frac{\delta H}{\delta \omega} = u$, so $\Pperp u = 0$. Thus $B(F, H) = 0$. Integrating the remaining terms by parts recovers the standard vorticity transport equation. Setting $F = H$ yields the energy balance.
\end{proof}

\section{The Frohmanian Vorticity-Alignment Tether Kernel and Desingularization}

Define the unit vorticity direction field $\xi(x,t) := \omega(x,t) / \abs{\omega(x,t)}$ on $\Omega_+(t) := \{ x \in \T : \omega(x,t) \neq 0 \}$.

\begin{definition}[Frohmanian Vorticity-Alignment Tether Kernel $\mathcal{K}_{\mathrm{align}}$]\label{def:align_kernel}
Differentiating the periodic Biot--Savart representation yields the principal-value representation of the symmetric strain tensor:
\begin{equation}\label{eq:strain_BS}
S(x) = \mathrm{P.V.} \int_{\T} \mathcal{M}(\widehat{r}) \omega(y) \, dy + \mathcal{M}_{\mathrm{per}}(\omega)(x),
\end{equation}
where $\left(\mathcal{M}(\widehat{r}) v\right)_{ij} := \frac{3}{8\pi \abs{r}^3} \bigl( \widehat{r}_i (\widehat{r} \times v)_j + \widehat{r}_j (\widehat{r} \times v)_i \bigr)$. Contracting twice with $\xi(x)$ defines
\begin{equation}\label{eq:tether_align_kernel}
\mathcal{K}_{\mathrm{align}}(x, y) := \frac{3}{4\pi \abs{x - y}^3} \bigl( \widehat{r} \cdot \xi(x) \bigr) \det\bigl( \widehat{r}, \xi(y), \xi(x) \bigr).
\end{equation}
\end{definition}

\begin{proposition}[Geometric Cancellation Identity]\label{prop:geom_cancellation}
At every $x \in \Omega_+(t)$,
\begin{equation}\label{eq:stretching_decomp}
S(x)\xi(x) \cdot \xi(x) = \mathrm{P.V.} \int_{\T} \mathcal{K}_{\mathrm{align}}(x, y) \abs{\omega(y)} \, dy + \mathcal{R}_{\mathrm{per}}(x),
\end{equation}
with $\norm{\mathcal{R}_{\mathrm{per}}}_{L^\infty} \le C \norm{\omega}_{L^2}$. Moreover, $\mathcal{K}_{\mathrm{align}}(x, y) = 0$ whenever $\xi(y) = \pm \xi(x)$, and
\begin{equation}\label{eq:kernel_bound}
\abs{\mathcal{K}_{\mathrm{align}}(x, y)} \le \frac{3}{4\pi \abs{x - y}^3} \abs{\xi(y) \times \xi(x)} = \frac{3}{4\pi \abs{x - y}^3} \abs{\sin \theta(x, y)}.
\end{equation}
\end{proposition}

\begin{proof}
Contracting the Biot--Savart matrix kernel with $\xi_i(x)\xi_j(x)$ and using $\omega(y) = |\omega(y)|\xi(y)$ produces the triple-product form. Cauchy--Schwarz on the unit vectors yields the sine bound; the kernel vanishes on parallel directions by the triple product.
\end{proof}

\begin{corollary}[Constantin--Fefferman Spherical-Shell Desingularization]\label{cor:CF_bound}
Suppose there exist $\rho \in (0, \pi)$ and $K_\theta > 0$ such that $|\xi(y) \times \xi(x)| \le K_\theta |x - y|$ whenever $|x - y| \le \rho$ and $x,y \in \Omega_+(t)$. Then the kernel desingularizes to $O(|x-y|^{-2})$, and
\begin{equation}\label{eq:CF_desing}
\abs{S(x)\xi(x) \cdot \xi(x)} \le 3 K_\theta \rho \norm{\omega}_{L^\infty(\T)} + \sqrt{\frac{3}{4\pi \rho^3}} \norm{\omega}_{L^2(\T)} + C_{\mathrm{per}} \norm{\omega}_{L^1(\T)}.
\end{equation}
\end{corollary}

\begin{proof}
Split the integral into the near field $|x-y|\le\rho$ and the far field. On the near field the Lipschitz coherence upgrades the kernel to $O(r^{-2})$; the spherical shell integral becomes
\[
\int_0^\rho \frac{3K_\theta}{4\pi r^2} \cdot 4\pi r^2 \, dr = 3K_\theta\rho.
\]
On the far field Cauchy--Schwarz with the integrable tail of $r^{-6}$ produces the $O(\rho^{-3/2})$ term. Adding the smooth periodic remainder finishes the estimate.
\end{proof}

\section{Conditional Global Regularity Theorem and BKM Continuation}

For $p\ge 2$ the $L^p$ vorticity functionals satisfy the differential inequality obtained by testing the evolution law (using $B(F_p,H)\equiv 0$):
\begin{equation}\label{eq:Sp_riccati}
\frac{d}{dt} \norm{\omega}_{L^p} \le \norm{S}_{L^\infty} \norm{\omega}_{L^p}.
\end{equation}
Passing to $p\to\infty$ yields $\frac{d^+}{dt}M(t)\le\norm{S(t)}_{L^\infty}M(t)$.

\begin{hypothesis}[Critical Strain Integrability --- Form B]\label{hyp:B}
Along the maximal local Kato solution, for every finite $T\in(0,T^*)$ there exists $C_* = C_*(\norm{u_0}_{H^s},\nu,T)<\infty$ such that
\begin{equation}\label{eq:hyp_B}
\int_0^T \norm{S(t)}_{L^\infty(\T)}\, dt \le C_*.
\end{equation}
\end{hypothesis}

\begin{theorem}[Conditional Global Regularity]\label{thm:main}
If Hypothesis~\ref{hyp:B} holds, then $T^*=+\infty$ and $u$ extends to a unique global solution that is smooth for positive times, $u\in C^\infty(\T\times(0,\infty))$. Explicitly,
\[
\sup_{t\in[0,T]} \norm{\omega(t)}_{L^\infty} \le \norm{\omega_0}_{L^\infty}\exp(C_*).
\]
\end{theorem}

\begin{proof}
Gronwall's lemma applied to the differential inequality on any finite interval $[0,T]$ gives the exponential bound on $M(t)$. The Kato commutator estimate together with the Beale--Kato--Majda logarithmic inequality then keeps $\norm{u}_{H^s}$ bounded, contradicting maximality of a finite $T^*$. Parabolic regularity upgrades the solution to $C^\infty(\T\times(0,\infty))$ for $t>0$.
\end{proof}

\begin{corollary}[One-directional BKM continuation]\label{cor:bkm_equiv}
For the maximal local Kato solution the following one-directional implications hold:
\begin{enumerate}
\item Hypothesis~\ref{hyp:B} $\Rightarrow$ $\int_0^{T}\norm{\omega}_{L^\infty}\,dt<\infty$ on every finite horizon $T$, whence $T^*=+\infty$ and $u\in C^\infty(\T\times(0,\infty))$;
\item conversely, any globally smooth solution satisfies Hypothesis~\ref{hyp:B} on every finite interval (continuous strain on compact time intervals).
\end{enumerate}
In particular, $\int_0^{T}\norm{\omega}_{L^\infty}\,dt<\infty\Rightarrow T^{*}>T$ (classical BKM continuation, one-directional).
\end{corollary}

\begin{proof}
(i) follows from the exponential bound of Theorem~\ref{thm:main} together with the Beale--Kato--Majda continuation theorem and parabolic bootstrap for $t>0$. (ii) is immediate from continuous Sobolev embedding on compact time intervals.
\end{proof}
