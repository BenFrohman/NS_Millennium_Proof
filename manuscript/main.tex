\section{Introduction}\label{sec:intro}

The incompressible Navier--Stokes equations on $\T=\R^3/\mathbb{Z}^3$ (or on $\R^3$ with finite energy) are
\begin{equation}\label{eq:NS}
\partial_t u+(u\cdot\nabla)u+\nabla p=\nu\Delta u,\qquad \nabla\cdot u=0,\qquad u(0)=u_0,
\end{equation}
with $\nu>0$. Taking the curl yields the vorticity transport equation for $\omega=\nabla\times u$:
\begin{equation}\label{eq:vort}
\partial_t\omega+(u\cdot\nabla)\omega=(\omega\cdot\nabla)u+\nu\Delta\omega .
\end{equation}
By the Beale--Kato--Majda theorem \cite{BKM1984}, a smooth solution can be continued past $T$ whenever $\int_0^T\|\omega(t)\|_{L^\infty}\,dt<\infty$. A uniform ceiling on $M(t)=\|\omega(t)\|_{L^\infty}$ therefore implies global regularity. Obtaining such a bound for arbitrary large smooth data is the core difficulty of the Clay Millennium Problem \cite{Fefferman2006}: the vortex-stretching term $(\omega\cdot\nabla)u$ is supercritical with respect to the natural parabolic scaling
\begin{equation}\label{eq:scaling}
u_\lambda(x,t)=\lambda u(\lambda x,\lambda^2t),\qquad \omega_\lambda(x,t)=\lambda^2\omega(\lambda x,\lambda^2 t),
\end{equation}
which leaves \eqref{eq:NS} invariant at fixed $\nu$.

The Frohmanian programme approaches this problem through a two-layer architecture:
\begin{enumerate}
\item \textbf{Geometric layer (\texttt{ArnoldGeometric.lean}, \texttt{SymplecticTether.lean}, \texttt{Uniqueness.lean}, \texttt{ForMathlib/Projection.lean}):} Starting from Arnold's coadjoint orbit of volume-preserving diffeomorphisms \cite{Arnold1966,ArnoldKhesin1998}, we introduce the \textbf{Frohmanian Symplectic Tether} $\Tf$—realized concretely via the $L^2$-projected quadratic kernel \lean{TetherKernel} with universal Calderón--Zygmund coupling $\kappa=\CZ(3)=3/2$. By projecting functional derivatives orthogonally to the velocity field $u=\delta H/\delta\omega$ via the Gram--Schmidt projector $\Pi_u$ (\lean{Pi_u}), the tether satisfies exact degeneracy $B(F,H)=0$ on the kinetic-energy Hamiltonian $H$ (\lean{tetherKernel_degenerates_on_kinetic_energy}), leaving the reversible Euler vector field unmodified while supplying controllable negative quadratic feedback $B(F,F)=-\kappa\int|\omega|^2|\Pi_u\delta F/\delta\omega|^2$ (\lean{ProducesControllableNegativeFeedback}).
\item \textbf{Analytic layer (\texttt{NS\_Equations.lean}, \texttt{TetheredLyapunov.lean}, \texttt{IndependentMajorant.lean}, \texttt{CubicSourceCertificate.lean}, \texttt{AnalyticPipeline.lean}, \texttt{ClosureBlueprint.lean}, \texttt{GlobalRegularity.lean}, \texttt{ClayStatements.lean}):} Using the quartic Lyapunov functional $S_\varepsilon(t)$ weighted by the transported enstrophy-accumulation scalar $\varphi$ (\lean{LyapunovS}, \lean{IsEnstrophyAccumulation}), the programme links the conjectured cubic tether damping rate $M'(t)\le \CZ M(t)^2-\kappa'' M(t)^3$ (\lean{FrohmanianTetherRate}) to the autonomous comparison ODE $y'=Cy^2-\kappa''y^3$ (\lean{comparison_ode_stability}, \lean{comparison_majorant_global_bound}), yielding the uniform ceiling $M(t)\le\max(M(0),\CZ/\kappa'')$ (\lean{riccati_ceiling_of_vorticity_di}) and closing global regularity via BKM continuation and uniform Kato restart.
\end{enumerate}

\begin{theorem}[Conditional Global Regularity; \lean{frohmanian_tether_theorem}, \lean{global_regularity_for_NS}]\label{thm:main}
Let $\nu>0$ and let $u_0$ be smooth (\lean{ContDiff ℝ ⊤}), divergence-free, and of finite $L^2$ energy. Assume within the class of physical solutions (\lean{IsPhysicalNSSolution}):

\section{Setting and the Physical Solution Class}\label{sec:setting}

Write $M(t)=\|\omega(t)\|_{L^\infty}$ (Lean: \lean{vorticity_sup_norm (vorticity (u t))}). In the Lean formalization, the pointwise Navier--Stokes equation is \lean{NS_PDE u p ν}. Because Mathlib's \lean{deriv} returns $0$ at points where a function of time is not differentiable, a spatially smooth field with jump discontinuities in time would trivially satisfy \lean{NS_PDE} unless time-differentiability and energy control are enforced.

We therefore define the physical solution class \lean{IsPhysicalNSSolution} in \lean{NS_Equations.lean}:
\leanfile{IsPhysicalNSSolution}
A time-dependent velocity field $u$ belongs to \lean{IsPhysicalNSSolution} if and only if for all $t\ge0$:
\begin{enumerate}
\item $s\mapsto u(s,x)$ is pointwise differentiable at $t$ for every $x\in\R^3$;
\item $x\mapsto u(t,x)$ is continuous and has finite kinetic energy $\int_{\R^3}\|u(t,x)\|^2\,dx<\infty$;
\item $u$ obeys the classical Leray $L^2$ energy inequality $\int_{\R^3}\|u(t,x)\|^2\,dx\le\int_{\R^3}\|u(0,x)\|^2\,dx$;
\item $u(t)=u(0)$ for $t\le0$ (initial-value convention on $(-\infty,0]$).
\end{enumerate}

The classical PDE inputs enter as explicit, transparent hypotheses:
\begin{itemize}
\item Kato local existence on physical solutions (\lean{KatoLocalWitness}, carrying \lean{phys : IsPhysicalNSSolution u}) \cite{Kato1984,Temam1995};
\item Beale--Kato--Majda continuation and parabolic smoothing (\lean{bkm_regularity_pipeline}) \cite{BKM1984,ConstantinFoias1988};
\item Energy-Grönwall uniqueness of smooth physical solutions (\lean{uniqueness_of_existence_times_of_energy_gronwall}) \cite{Temam1995}.
\end{itemize}

\section{Geometric Layer: The Frohmanian Symplectic Tether}\label{sec:geom}

\subsection{The Classical Arnold Lie--Poisson Bracket}
On the coadjoint orbit $\mathcal O$ of the group $\mathrm{SDiff}(\T)$ of volume-preserving diffeomorphisms, the classical Kirillov--Kostant--Souriau / Arnold Lie--Poisson bracket \cite{Arnold1966,ArnoldKhesin1998} is
\[
\{F,G\}_{\mathrm{cl}}(\omega)=\int_{\T}\omega\cdot\Bigl(\frac{\delta F}{\delta\omega}\times\frac{\delta G}{\delta\omega}\Bigr)\,dV .
\]
With kinetic-energy Hamiltonian $H(\omega)=\tfrac12\int|u|^2\,dV$ (where $\delta H/\delta\omega=u=\mathrm{BS}(\omega)$ is the Biot--Savart velocity reconstruction, formalized in \lean{ArnoldGeometric.lean}), the Hamiltonian vector field of $\{F,H\}_{\mathrm{cl}}$ reproduces the ideal 3D Euler vorticity equation ($\nu=0$ in \eqref{eq:vort}; Lean: \lean{classical_bracket_reproduces_Euler}).

\subsection{Route A: The Projected Tether Kernel and Degeneracy}
\begin{definition}[Frohmanian tether kernel; Lean \lean{TetherKernel}, \lean{Pi_u}]\label{def:tether}
For a velocity field $u=\mathrm{BS}(\omega)$ with $\|u\|_{L^2}\neq0$, let $\Pi_u=P^\perp_u$ be the $L^2$-orthogonal Gram--Schmidt projection onto the complement of $\operatorname{span}\{u\}$:
\[
\Pi_u v=v-\frac{\langle v,u\rangle_{L^2}}{\|u\|_{L^2}^2}\,u
\]
(Lean: \lean{ForMathlib.Pi_u}, with $\Pi_u v=v$ when $\|u\|_{L^2}=0$). With universal 3D Calderón--Zygmund constant $\kappa=\CZ(3)=\frac{3}{8\pi}\cdot 4\pi=\frac{3}{2}$ (Lean: \lean{kappa_eq_three_halves} in \lean{ForMathlib/NS/Tether.lean}), define the Frohmanian tether kernel
\[
B(F,G)(\omega)=-\kappa\int_{\T}|\omega(x)|^2\,\Bigl\langle \Pi_u\Bigl(\frac{\delta F}{\delta\omega}\Bigr)(x),\,\Pi_u\Bigl(\frac{\delta G}{\delta\omega}\Bigr)(x)\Bigr\rangle\,dV .
\]
The full tethered bracket is $\{F,G\}_{\Tf}=\{F,G\}_{\mathrm{cl}}+B(F,G)$.
\end{definition}

\begin{proposition}[Strict Degeneracy on Kinetic Energy; proved, \lean{tetherKernel_degenerates_on_kinetic_energy}, \lean{tetherKernel_C2}]\label{prop:degen}
Whenever $\delta H/\delta\omega=u$, $\nabla\cdot u=0$, and $u\in L^2$, one has $\Pi_u u=0$ (\lean{projection_orthogonal_to_u}) and hence
\[
B(F,H)(\omega)=0\qquad\text{for every functional }F.
\]
\end{proposition}
\begin{proof}
If $\|u\|_{L^2}\ne0$, $\Pi_u u=u-(\langle u,u\rangle_{L^2}/\|u\|_{L^2}^2)u=0$, so the projected factor $\Pi_u(\delta H/\delta\omega)$ vanishes pointwise. If $\|u\|_{L^2}=0$, $u=0$ a.e.\ and $\Pi_u 0=0$ (\lean{Pi_u_zero}). Thus $\{F,H\}_{\Tf}=\{F,H\}_{\mathrm{cl}}$ for all $F$: the reversible Euler vector field generated by $H$ is preserved unmodified.
\end{proof}

\subsection{Symmetry vs.\ Antisymmetry and Dual Uniqueness Theorems}
As an explicit bilinear integral kernel, $B(F,G)$ in Definition~\ref{def:tether} is \emph{symmetric} in $(F,G)$ (\lean{tetherKernel_comm}, \lean{tetherKernel_symmetric}), acting as a weighted, $L^2$-projected metric pairing of metriplectic type \cite{Morrison1986}. In \lean{Uniqueness.lean}, we formalize and prove \textbf{both} uniqueness characterizations of \lean{TetherKernel}:
\begin{enumerate}
\item \textbf{Symmetric metric-tether uniqueness (\lean{uniqueness_of_minimal_symmetric_tether}):} Among all \emph{symmetric} bilinear kernels $B'(\omega,F,G)=B'(\omega,G,F)$ satisfying coadjoint invariance (C1), kinetic-energy degeneracy (C2), controllable negative feedback (C3), lowest-order canonical density (\lean{IsLowestOrderTerm}), and quadratic saturation (\lean{SaturatesTetherQuadratic}), $B'$ is uniquely equal to \lean{TetherKernel}. Moreover, \lean{TetherKernel} itself satisfies the symmetry hypothesis (\lean{tetherKernel_symmetric}).
\item \textbf{Antisymmetric uniqueness (\lean{uniqueness_of_minimal_tether}):} Preserves the original antisymmetric binder $\forall\omega F G,\ B'(\omega,F,G)=-B'(\omega,G,F)$ alongside (C1)--(C3), \lean{IsLowestOrderTerm}, and \lean{SaturatesTetherQuadratic}, proving $B'=\text{\lean{TetherKernel}}$ on the locus where those hypotheses hold.

\begin{lemma}[Stretching at a spatial maximum; proved, \lean{stretching_inner_le_at_max}]\label{lem:localmax}
Let $x$ satisfy $\|\omega(x)\|=\|\omega\|_{L^\infty}$ and suppose $\|\nabla u(x)\|\le\CZ\|\omega\|_{L^\infty}$ (\lean{hCZ}). Then

\langle\omega(x),(\omega\cdot\nabla)u(x)\rangle\le\CZ\,\|\omega\|_{L^\infty}^3 .

\end{lemma}

\begin{lemma}[Transport and viscosity at a spatial maximum; proved, \lean{time_pairing_le_stretching_at_max}]\label{lem:transport}

\langle\omega,\partial_t\omega\rangle(x)\le\langle\omega,(\omega\cdot\nabla)u\rangle(x).
\]
\end{lemma}

y'(s)\le C\,y(s)^2-\kappa''\,y(s)^3\qquad(s\ge0).

Then for every $t\ge0$,
\[
y(t)\le\max\Bigl(y(0),\ \frac{C}{\kappa''}\Bigr).
\]
\end{lemma}

\end{lemma}

\begin{proposition}[Cubic rate $\Rightarrow$ uniform vorticity ceiling; proved, \lean{riccati_ceiling_of_vorticity_di}]\label{prop:ceiling}
If $M(t)=\|\omega(t)\|_{L^\infty}$ is differentiable on $[0,\infty)$ and satisfies $M'(t)\le \CZ M(t)^2-\kappa''M(t)^3$, then
\[
\sup_{t\ge0}M(t)\le\max\Bigl(M(0),\frac{\CZ}{\kappa''}\Bigr).
\]
\end{proposition}

\subsection{Resolution of the Unpatched Time-Derivative Vacuity and Proof of Joint Satisfiability}\label{sec:vacuity}
